function uid(){
  return (crypto.randomUUID ? crypto.randomUUID() : Date.now()+"-"+Math.random().toString(16).slice(2));
}
function esc(v=""){
  return String(v).replace(/[&<>"']/g, m => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#039;'}[m]));
}
function fmtDate(value){
  if(!value) return "—";
  const d = new Date(value + "T00:00:00");
  return d.toLocaleDateString(undefined,{year:"numeric",month:"short",day:"numeric"});
}
function daysBetween(date){
  if(!date) return null;
  const start = new Date(date + "T00:00:00");
  return Math.max(0, Math.floor((new Date()-start)/86400000));
}
function humanSince(date){
  const days = daysBetween(date);
  if(days === null) return "—";
  if(days < 31) return days + (days===1 ? " day" : " days");
  if(days < 365){
    const months = Math.floor(days/30.44);
    return months + (months===1 ? " month" : " months");
  }
  const years = Math.floor(days/365.25);
  const months = Math.floor((days-years*365.25)/30.44);
  return years + "y" + (months ? " " + months + "m" : "");
}
function render(){
  const q = els.search.value.trim().toLowerCase();
  const loc = els.locationFilter.value;
  let list = plants.filter(p => {
    const hay = [p.type,p.genus,p.cultivar,p.subgenre,p.nickname,p.location,p.source,p.price,p.rarity,p.potSize,p.origin,p.growthHabit,p.medium,p.light,p.support,p.health,p.notes].join(" ").toLowerCase();
    return (!q || hay.includes(q)) && (!loc || p.location === loc);
  });

  const sort = els.sortBy.value;
  list.sort((a,b)=>{
    if(sort==="type") return (a.type||"").localeCompare(b.type||"");
    if(sort==="acquired_desc") return (b.dateAcquired||"").localeCompare(a.dateAcquired||"");
    if(sort==="repot_oldest") return (a.lastRepot||"9999").localeCompare(b.lastRepot||"9999");
    if(sort==="pot") return (a.potSize||"").localeCompare(b.potSize||"", undefined,{numeric:true});
    return (a.nickname||a.type||"").localeCompare(b.nickname||b.type||"");
  });

  els.rows.innerHTML = list.map(p=>{
    const d = daysBetween(p.lastRepot);
    const cls = d !== null && d >= 365 ? "badge warn" : "badge";
    return `<tr>
      <td>${p.photo ? `<img class="plant-photo" src="${p.photo}" alt="${esc(p.nickname || p.type || 'Plant')}">` : `<div class="photo-placeholder">🌱</div>`}</td>
      <td><button class="icon-btn favorite-btn" onclick="toggleFavorite('${p.id}')" title="Favorite">${p.favorite ? "★" : "☆"}</button></td>
      <td class="plant-name">${esc(p.type)||"—"}${p.notes ? `<div class="details">${esc(p.notes).slice(0,80)}${p.notes.length>80?"…":""}</div>` : ""}</td>
      <td>${esc(p.genus)||"—"}</td>
      <td>${esc(p.cultivar)||"—"}</td>
      <td>${esc(p.subgenre)||"—"}</td>
      <td class="nickname">${esc(p.nickname)||"—"}</td>
      <td>${fmtDate(p.dateAcquired)}</td>
      <td>${esc(p.location)||"—"}</td>
      <td>${esc(p.source)||"—"}</td>
      <td>${esc(p.price)||"—"}</td>
      <td>${esc(p.rarity)||"—"}</td>
      <td>${p.quantity || 1}</td>
      <td><span class="badge health-${esc(p.health||"Good")}">${esc(p.health||"Good")}</span></td>
      <td>${esc(p.potSize)||"—"}</td>
      <td>${fmtDate(p.lastRepot)}</td>
      <td><span class="${cls}">${humanSince(p.lastRepot)}</span></td>
      <td>${fmtDate(p.lastWatered)}</td>
      <td>${fmtDate(p.lastFertilized)}</td>
      <td><div class="row-actions">
        <button class="icon-btn secondary" onclick="quickLog('${p.id}','Watered')">💧 Water</button>
        <button class="icon-btn secondary" onclick="quickLog('${p.id}','Fertilized')">🌱 Feed</button>
        <button class="icon-btn secondary" onclick="viewProfile('${p.id}')">Profile</button>
        <button class="icon-btn secondary" onclick="editPlant('${p.id}')">Edit</button>
        <button class="icon-btn danger" onclick="deletePlant('${p.id}')">Delete</button>
      </div></td>
    </tr>`;
  }).join("");

  els.empty.style.display = list.length ? "none" : "block";
  updateStats();
  updateLocations();
}
function updateStats(){
  document.getElementById("totalPlants").textContent = plants.length;
  document.getElementById("locationsCount").textContent =
    new Set(plants.map(p=>p.location).filter(Boolean)).size;
  document.getElementById("recentCount").textContent =
    plants.filter(p=>{ const d=daysBetween(p.dateAcquired); return d!==null && d<=30; }).length;
  document.getElementById("repotDueCount").textContent =
    plants.filter(p=>{ const d=daysBetween(p.lastRepot); return d!==null && d>=365; }).length;
}
function updateLocations(){
  const current = els.locationFilter.value;
  const locations = [...new Set(plants.map(p=>p.location).filter(Boolean))].sort();
  els.locationFilter.innerHTML = '<option value="">All locations</option>' +
    locations.map(x=>`<option ${x===current?"selected":""} value="${esc(x)}">${esc(x)}</option>`).join("");
}

window.toggleFavorite = id => {
  const p=plants.find(x=>x.id===id); if(!p)return;
  p.favorite=!p.favorite; savePlants(); render();
};
window.quickLog = (id, action) => {
  const p=plants.find(x=>x.id===id); if(!p)return;
  const today=new Date().toISOString().slice(0,10);
  p.history=p.history||[];
  p.history.unshift({date:today,action});
  if(action==="Watered") p.lastWatered=today;
  if(action==="Fertilized") p.lastFertilized=today;
  savePlants(); render();
};
window.editPlant = id => openForm(plants.find(p=>p.id===id));
window.deletePlant = id => {
  const plant = plants.find(p=>p.id===id);
  if(confirm(`Delete ${plant?.nickname || plant?.type || "this plant"}?`)){
    plants = plants.filter(p=>p.id!==id);
    savePlants(); render();
  }
};

document.getElementById("exportBtn").addEventListener("click",()=>{
  const blob = new Blob([JSON.stringify(plants,null,2)],{type:"application/json"});
  const url = URL.createObjectURL(blob);
  const a=document.createElement("a");
  a.href=url;
  a.download="plant-collection-backup.json";
  a.click();
  URL.revokeObjectURL(url);
});

document.getElementById("importFile").addEventListener("change",async e=>{
  const file=e.target.files[0];
  if(!file) return;
  try{
    const incoming=JSON.parse(await file.text());
    if(!Array.isArray(incoming)) throw new Error();
    plants=incoming;
    savePlants(); render();
    alert("Plant collection imported successfully.");
  }catch{
    alert("That file doesn't look like a valid plant tracker backup.");
  }
  e.target.value="";
});
