img,0,0,w,h);
        resolve(canvas.toDataURL("image/jpeg",quality));
      };
      img.onerror=reject;img.src=reader.result;
    };
    reader.onerror=reject;reader.readAsDataURL(file);
  });
}

function renderTimeline(){
  const p = profilePlant();
  const timeline = [...(p?.history || [])].sort((a,b)=>(b.date||"").localeCompare(a.date||""));
  const el = document.getElementById("profileTimeline");
  if(!timeline.length){
    el.innerHTML = `<div class="empty">No care events yet. Use the quick buttons or add one above.</div>`;
    return;
  }
  el.innerHTML = timeline.map((item,i)=>`
    <div class="timeline-item">
      <div class="timeline-date">${item.date ? fmtDate(item.date) : "Undated"}</div>
      <div>
        <div class="timeline-action">${esc(item.action || "Activity")}</div>
        ${item.note ? `<div class="timeline-note">${esc(item.note)}</div>` : ""}
      </div>
      <button class="icon-btn danger" type="button" onclick="removeLog(${i})">Delete</button>
    </div>`).join("");
}

window.removeLog = index => {
  const p = profilePlant(); if(!p) return;
  const sorted = [...(p.history || [])].sort((a,b)=>(b.date||"").localeCompare(a.date||""));
  const target = sorted[index];
  const originalIndex = (p.history || []).findIndex(x => x === target);
  if(originalIndex >= 0) p.history.splice(originalIndex,1);
  savePlants(); renderProfile();
};

document.getElementById("addLogBtn").addEventListener("click", ()=>{
  const p = profilePlant(); if(!p) return;
  const date = document.getElementById("logDate").value || new Date().toISOString().slice(0,10);
  const action = document.getElementById("logAction").value;
  const note = document.getElementById("logNote").value.trim();
  p.history = p.history || [];
  p.history.unshift({date,action,note});

  if(action === "Watered") p.lastWatered = date;
  if(action === "Fertilized") p.lastFertilized = date;
  if(action === "Repotted") p.lastRepot = date;

  savePlants();
  document.getElementById("logNote").value = "";
  renderProfile();
});

/* Plantdex permanent profile links, NFC + QR support */
function plantDisplayId(p){
  let h = 2166136261;
  const s = String(p?.id || "plant");
  for(let i=0;i<s.length;i++){
    h ^= s.charCodeAt(i);
    h = Math.imul(h,16777619);
  }
  return "PD-" + (h >>> 0).toString(36).toUpperCase().padStart(7,"0").slice(-7);
}

function plantProfileUrl(p){
  const u = new URL(window.location.href);
  u.search = "";
  u.hash = "";
  u.searchParams.set("plant", p.id);
  return u.toString();
}

function renderNfcCard(){
  const p = profilePlant();
  if(!p) return;
  const grid = document.querySelector(".profile-grid");
  if(!grid) return;
  let card = document.getElementById("nfcProfileCard");
  if(!card){
    card = document.createElement("div");
    card.id = "nfcProfileCard";
    card.className = "profile-card";
    grid.appendChild(card);
  }
  const url = plantProfileUrl(p);
  const qr = `https://api.qrserver.com/v1/create-qr-code/?size=220x220&qzone=4&data=${encodeURIComponent(url)}`;
  card.innerHTML = `
    <h3>🏷️ NFC / QR Profile</h3>
    <div style="display:grid;grid-template-columns:minmax(150px,220px) 1fr;gap:18px;align-items:center">
      <img src="${qr}" width="220" height="220" alt="QR code for ${esc(p.nickname || p.type || 'plant')} profile" style="width:100%;max-width:220px;border-radius:12px;border:1px solid var(--line);background:white;padding:8px">
      <div>
        <div style="font-size:.8rem;color:var(--muted);font-weight:800;text-transform:uppercase;letter-spacing:.05em">Plant ID</div>
        <div style="font-size:1.35rem;font-weight:850;margin:4px 0 14px">${plantDisplayId(p)}</div>
        <div style="font-size:.8rem;color:var(--muted);font-weight:800;text-transform:uppercase;letter-spacing:.05em">Permanent profile URL</div>
        <div style="word-break:break-all;margin:5px 0 12px">${esc(url)}</div>
        <div style="display:flex;gap:8px;flex-wrap:wrap">
          <button class="primary" type="button" onclick="copyPlantProfileUrl()">Copy NFC Link</button>
          <a class="secondary" href="${qr}" target="_blank" rel="noopener" style="display:inline-flex;align-items:center;text-decoration:none;border-radius:12px;padding:11px 15px;font-weight:700">Open QR</a>

}

window.copyPlantProfileUrl = async () => {
  const p = profilePlant(); if(!p) return;
  const url = plantProfileUrl(p);
  try{
    await navigator.clipboard.writeText(url);
    alert("Plant NFC link copied.");
  }catch{
    const input=document.createElement("textarea");
    input.value=url;document.body.appendChild(input);input.select();document.execCommand("copy");input.remove();
    alert("Plant NFC link copied.");
  }
};
