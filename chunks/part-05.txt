img,0,0,w,h);
        resolve(canvas.toDataURL("image/jpeg",quality));
      };
      img.onerror=reject;img.src=reader.result;
    };
    reader.onerror=reject;reader.readAsDataURL(file);
  });
}

let activeProfileId = null;

function profilePlant(){
  return plants.find(p => p.id === activeProfileId);
}

window.viewProfile = id => {
  activeProfileId = id;
  renderProfile();
  document.querySelector(".panel").style.display = "none";
  document.querySelector(".stats").style.display = "none";
  document.getElementById("profileShell").style.display = "block";
  window.scrollTo({top:0,behavior:"smooth"});
};

window.closeProfile = () => {
  activeProfileId = null;
  document.getElementById("profileShell").style.display = "none";
  document.querySelector(".panel").style.display = "block";
  document.querySelector(".stats").style.display = "grid";
};

  document.getElementById("profileHeader").innerHTML = `
    ${hero}
    <div>
      <button class="secondary profile-back" type="button" onclick="closeProfile()">← Back to Collection</button>
      <h2 class="profile-title">${esc(p.nickname || p.type || "Unnamed Plant")}</h2>

        ${p.rarity ? `<span class="badge">${esc(p.rarity)}</span>` : ""}
        ${p.health ? `<span class="badge health-${esc(p.health)}">${esc(p.health)}</span>` : ""}
        ${p.favorite ? `<span class="badge">★ Favorite</span>` : ""}
        ${p.location ? `<span class="badge">${esc(p.location)}</span>` : ""}
      </div>
      <div class="profile-actions">
        <button class="primary" type="button" onclick="editPlant('${p.id}')">Edit Plant</button>
        <button class="secondary" type="button" onclick="quickLog('${p.id}','Watered');renderProfile()">💧 Watered Today</button>
        <button class="secondary" type="button" onclick="quickLog('${p.id}','Fertilized');renderProfile()">🌱 Fertilized Today</button>

      </div>
    </div>`;

    ["Genus / Family", p.genus],
    ["Cultivar / Variety", p.cultivar],
    ["Sub Genre", p.subgenre],
    ["Quantity", p.quantity || 1],
    ["Date Acquired", fmtDate(p.dateAcquired)],
    ["Acquired From", p.source],
    ["Price Paid", p.price],
    ["Estimated Value", p.estimatedValue],
    ["Rarity", p.rarity],
    ["Propagation / Origin", p.origin],
    ["Growth Habit", p.growthHabit],
    ["Notes", p.notes]

    ["Pot Size", p.potSize],
    ["Potting Medium", p.medium],
    ["Support", p.support],
    ["Last Repot", fmtDate(p.lastRepot)],
    ["Since Repot", humanSince(p.lastRepot)],
    ["Last Watered", fmtDate(p.lastWatered)],
    ["Last Fertilized", fmtDate(p.lastFertilized)],

  renderTimeline();

  document.getElementById("logDate").value = new Date().toISOString().slice(0,10);
}

function renderGallery(){
  const p = profilePlant();
  const gallery = p?.gallery || [];
  const el = document.getElementById("profileGallery");
  if(!gallery.length){
    el.innerHTML = `<div class="empty">No growth photos yet. Add one above to start a visual timeline.</div>`;
    return;
  }
  el.innerHTML = gallery.map((g,i)=>`
    <div class="gallery-item">
      <img src="${g.src}" alt="Plant progress photo">
      <button class="gallery-remove" type="button" onclick="removeGalleryPhoto(${i})">Remove</button>
      <div class="gallery-date">${g.date ? fmtDate(g.date) : "Undated"}</div>
    </div>`).join("");
}

window.removeGalleryPhoto = index => {
  const p = profilePlant(); if(!p) return;
  p.gallery = p.gallery || [];
  p.gallery.splice(index,1);
  savePlants(); renderGallery();
};

document.getElementById("addGalleryPhotoBtn").addEventListener("click", async ()=>{
  const p = profilePlant();
  const file = document.getElementById("galleryPhoto").files[0];
  if(!p || !file) return;
  const src = await resizeImage(file, 1200, .8);
  p.gallery = p.gallery || [];
  p.gallery.unshift({
    src,
    date: document.getElementById("galleryDate").value || new Date().toISOString().slice(0,10),
    note: ""
  });
  if(!p.photo) p.photo = src;
  savePlants();
  document.getElementById("galleryPhoto").value = "";
  renderProfile();
});

function renderTimeline(){
  const p = profilePlant();
  const timeline = [...(p?.history || [])].sort((a,b)=>(b.date||"").localeCompare(a.date||""));
  const el = document.getElementById("profileTimeline");
  if(!timeline.length){
    el.innerHTML = `<div class="empty">No care events yet. Use the quick buttons or add one above.</div>`;
    return;
  }
  el.innerHTML = timeline.map((item,i)=>`
    <div class="timeline-item">
      <div class="timeline-date">${item.date ? fmtDate(item.date) : "Undated"}</div>
      <div>
        <div class="timeline-action">${esc(item.action || "Activity")}</div>
        ${item.note ? `<div class="timeline-note">${esc(item.note)}</div>` : ""}
      </div>
      <button class="icon-btn danger" type="button" onclick="removeLog(${i})">Delete</button>
    </div>`).join("");
}

window.removeLog = index => {
  const p = profilePlant(); if(!p) return;
  const sorted = [...(p.history || [])].sort((a,b)=>(b.date||"").localeCompare(a.date||""));
  const target = sorted[index];
  const originalIndex = (p.history || []).findIndex(x => x === target);
  if(originalIndex >= 0) p.history.splice(originalIndex,1);
  savePlants(); renderProfile();
};

document.getElementById("addLogBtn").addEventListener("click", ()=>{
  const p = profilePlant(); if(!p) return;
  const date = document.getElementById("logDate").value || new Date().toISOString().slice(0,10);
  const action = document.getElementById("logAction").value;
  const note = document.getElementById("logNote").value.trim();
  p.history = p.history || [];
  p.history.unshift({date,action,note});

  if(action === "Watered") p.lastWatered = date;
  if(action === "Fertilized") p.lastFertilized = date;
  if(action === "Repotted") p.lastRepot = date;

  savePlants();
  document.getElementById("logNote").value = "";
  renderProfile();
});
